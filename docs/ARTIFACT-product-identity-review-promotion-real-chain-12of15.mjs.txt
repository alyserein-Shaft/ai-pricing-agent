const actor=(permission="Library Manager")=>({id:"user-a",permission,role:permission});
const req=(operation,body)=>new Request(`http://local/api/product-identities/identity-a/${operation}`,{method:"POST",headers:{"content-type":"application/json"},body:JSON.stringify(body)});
const reviewBody=(overrides={})=>({manufacturer:"Johnson Controls",unit:"RL",reason:"Verified against explicit supplier quotation evidence.",evidence:{facts:["fact-a"]},idempotencyKey:"review-key",expectedVersion:1,...overrides});
const promoteBody=(overrides={})=>({reason:"Promote reviewed identity as additive organization knowledge.",idempotencyKey:"promote-key",expectedVersion:2,...overrides});
const performReview=(DB,overrides={})=>reviewProductIdentity(req("review",reviewBody(overrides)),{DB},actor("Library Reviewer"),{id:"org-a"},"identity-a");

const performPromotion=(DB,overrides={},user=actor())=>promoteProductIdentity(req("promote",promoteBody(overrides)),{DB},user,{id:"org-a"},"identity-a");
